首先是数据结构类型上，最基础的、不含迭代器的结构有 queue, stack ，这两者作为初学者最开始碰到的容器，特性非常简单。queue是FIFO（先进先出）、有两端、保序；stack是LIFO（后进先出）、只有一端、逆序。

对于栈与队列，接口也比较简单，易于记忆，双手可数：empty,top,back,front,push,emplace,pop———没了。接口作用也可以望文生义———但是要注意一点，C++中要使用top/front和pop搭配才相当于python中的pop，top/front/back都是访问，pop才是弹出。

然后，由于栈与队列的操作大多数是对string进行操作，所以大多数题目相对简单（所以刷题的时候顿感 just so so～），当然有兴趣的也可以学我用双栈法完成 Leetcode160———单纯地为了验证自己的想法，误打误撞用到了栈（srds并无必要）。

二叉树的类型比较简单，但是很多时候并无必要分辨，我们着重要讲的是搜索树。其中涉及到四种思想：递归、迭代、深度优先搜索（$Depth-First search$，也作$DFS$）、广度优先搜索（$Breadth-First search$，也作$BFS$）。

以下是我根据不同的搜索顺序和访问顺序的不同组合，完成的层序遍历：

```cpp
class BFSQueue {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        queue<TreeNode*> que;                       // 层序遍历使用队列
        if (root != nullptr) que.push(root);        // 根节点入队
        vector<vector<int>> result;                 // 结果集

        while (!que.empty()) {                  // 队列不为空，说明还有节点没有遍历完
            int levelSize = static_cast<int>(que.size());   // 当前层的节点数
            vector<int> vec;

// 这里一定要使用固定大小 levelSize，不要直接用 que.size()，因为它在循环中会不断变化

            for (int i = 0; i < levelSize; ++i) {   // 遍历当前层的所有节点
                TreeNode* node = que.front();       // 获取队首节点
                que.pop();                          // 队首节点出队
                vec.push_back(node->val);          // 将当前节点的值加入当前层的结果集

                if (node->left)  que.push(node->left);      // 左子节点入队
                if (node->right) que.push(node->right);     // 右子节点入队
            }
            result.push_back(std::move(vec));       // 本层的结果集收进总结果集
        }
        return result;                              // 返回结果集
    }
};
```

```cpp
class BFSRecursive {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        vector<vector<int>> result;                 // 结果集
        if (root != nullptr) bfsNextLevel({root}, result);     
        // 从"只有根节点"这一层开始
        return result;                              // 返回结果集
    }

private:
    /* 辅助函数：处理"当前层"，并把下一层交给递归 */
    void bfsNextLevel(const vector<TreeNode*>& level, vector<vector<int>>& result) {
        if (level.empty()) return;                  // 本层没有节点，递归结束

        vector<int> vec;                            // 本层的结果集
        vector<TreeNode*> nextLevel;                // 下一层的节点列表

        for (TreeNode* node : level) {              // 依次处理本层的每一个节点
            vec.push_back(node->val);               // 将当前节点的值加入本层的结果集

            if (node->left)  nextLevel.push_back(node->left);       
            // 左子节点作为下一层
            if (node->right) nextLevel.push_back(node->right);      
            // 右子节点作为下一层
        }
        result.push_back(std::move(vec));           // 本层的结果集收进总结果集

        bfsNextLevel(nextLevel, result);            // 递归处理"下一层"
    }
};
```

```cpp
class DFSRecursive {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        vector<vector<int>> result;                 // 结果集
        dfs(root, 0, result);                       // 从根节点、第 0 层开始
        return result;                              // 返回结果集
    }

private:
    /* 辅助函数：先记下当前节点，再一路向左，回溯后向右
       —— 走访顺序是 DFS，节点落在哪一格靠的是深度 */
    void dfs(TreeNode* node, int depth, vector<vector<int>>& result) {
        if (node == nullptr) return;                // 空节点，递归结束

        if (depth == static_cast<int>(result.size()))
            result.push_back({});                   // 第一次到达该深度，新建结果集
        result[depth].push_back(node->val);         // 把当前节点放进对应深度的结果集

        dfs(node->left,  depth + 1, result);        // 先递归左子树
        dfs(node->right, depth + 1, result);        // 再递归右子树
    }
};
```

```cpp
class DFSStack {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        vector<vector<int>> result;                 // 结果集
        if (root == nullptr) return result;         // 空树直接返回

        stack<pair<TreeNode*, int>> st;             // 栈里存"节点 + 深度"
        st.push({root, 0});                         // 根节点入栈，深度为 0

        while (!st.empty()) {                       
        // 栈不为空，说明还有节点没有遍历完
            TreeNode* node = st.top().first;        // 获取栈顶节点
            int depth = st.top().second;            // 获取该节点的深度
            st.pop();                               // 栈顶出栈

        // 当前深度还没有对应的结果集，说明是第一次访问该深度的节点，需要创建一个新的结果集
            if (depth == static_cast<int>(result.size()))
                result.push_back({});               // 创建一个新的结果集

            result[depth].push_back(node->val);     
            // 将当前节点的值加入对应深度的结果集

            if (node->right) st.push({node->right, depth + 1});     // 先压右子节点
            if (node->left)  st.push({node->left,  depth + 1});     
            // 后压左子节点 → 下次先弹出左
        }
        return result;                              // 返回结果集
    }
};
```

其中每个题目的答案模板都出自以上四个方法，不同之处只在输出的答案。当我们在面对一个算法题时应该如何思考呢？

首先我们应该考虑：我需不需要跨节点？我需要求树深吗？我的返回类型是什么？我应该选择什么容器？从而我们就可以确定下我们是选择$DFS$还是$BFS$。选择不恰当的容器在进行计算的时候可能会出现容器溢出的报错，从而降低程序的可运行性。
